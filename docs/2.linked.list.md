
typedef struct Node
{
    int data;
    struct Node *next;
} Node;

Node * createSinglyLinkedList(){
    Node * head = (Node *)malloc(sizeof(Node));
    return head;
}

void addAtLast(Node* head, int val){
    if(head == NULL){
        head->data = val;
        return;
    }
    Node *temp = head;
    while(temp->next != NULL){
        temp = temp->next;
    }
    Node * n = (Node*)malloc(sizeof(Node));
    n->data = val;
    n->next = NULL;
}

void print(Node * head){
    Node * temp = head;
    while(temp->next){
        printf("%d ", temp->data);
        temp = temp->next;
    }
}
